cmake_minimum_required(VERSION 3.16)
project(privfim_nipp_upstream LANGUAGES CXX)
find_package(OpenMP REQUIRED)
set(CMAKE_CXX_STANDARD 14)
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(TFHE "${ROOT}/third_party/tfhe")
set(NIPP "${ROOT}/third_party/non-interactive-ppfim/ppfim_tfhe/src")
# 上游使用 <tfhe/tfhe.h>，本地 TFHE 尚未安装到系统目录。
file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/include")
file(CREATE_LINK "${TFHE}/src/include" "${CMAKE_BINARY_DIR}/include/tfhe" SYMBOLIC)
add_executable(nipp_upstream nipp_upstream.cpp "${NIPP}/cloud.cpp" "${NIPP}/user.cpp")
target_include_directories(nipp_upstream PRIVATE "${NIPP}" "${CMAKE_BINARY_DIR}/include")
target_compile_options(nipp_upstream PRIVATE -O3 -Wall -Wextra)
target_link_libraries(nipp_upstream PRIVATE "${TFHE}/build/libtfhe/libtfhe-spqlios-fma.so")
target_link_libraries(nipp_upstream PRIVATE OpenMP::OpenMP_CXX)
set_target_properties(nipp_upstream PROPERTIES BUILD_RPATH "${TFHE}/build/libtfhe")
